    \section{Richiami generali di calcolo differenziale}
        Nel seguito considereremo lo spazio vettoriale euclideo $\mathbb{R}^n$ munito del prodotto scalare standard

        $\forall x=(x^1,\ldots,x^n) \quad y=(y^1,\ldots,y^n) \in \mathbb{R}^n$ 
        \[ x \cdot y = \sum_{i=1}^n x^i y^i \]
        \[ \norm{x} = \sqrt{x \cdot x} = \sqrt{\sum_{i=1}^n (x^i)^2} \]
        \[d(x,y) = \norm{x-y}= \sqrt{\sum_{i=1}^n (x^i - y^i)^2} \]
        \begin{definizione}
        Sia $A \subset \mathbb{R}^n$ un insieme aperto e $F: A \to \mathbb{R}^m$ una funzione, si dice che $F$ è
        \textcolor{red}{differenziabile} in $x_0 \in A$ se $\exists L: \mathbb{R}^n \to \mathbb{R}^m$ applicazione linerare tale che
        \[ \lim_{x \to x_0} \frac{F(x) - F(x_0) - L(x-x_0)}{\norm{x-x_0}} = 0 \]
        $L$ se esiste è unica e prende il nome di \textcolor{red}{differenziale di $F$ in $x_0$} e si denota con \textcolor{red}{$L=\left(dF\right)_{x_0}$}

        Si dice che $F: A \to \mathbb{R}^m$ è \textcolor{red}{differenziabile} in $A$ se è differenziabile in ogni punto di $A$.
        \end{definizione}
        \begin{esempio} \textcolor{violet}{(Un caso molto semplice di applicazione differenziabile è un'applicazione lineare)}
        
        Se $L: \mathbb{R}^n \to \mathbb{R}^m$ è un'applicazione lineare allora $L$ è differenziabile e $\left(dL\right)_{x_0}=L$ per ogni $x_0 \in \mathbb{R}^n$
  
        Perchè $\displaystyle L(x)-L(x_0)-L(x-x_0)=0 \Rightarrow \lim_{x \to x_0} \frac{L(x)-L(x_0)-L(x-x_0)}{\norm{x-x_0}} = 0$
        \end{esempio}
        \begin{esempio} 
        \textcolor{violet}{(Un altro caso semplice di applicazione differenziabile è una traslazione di vettore $b$ che in generale non è lineare, lo è solo quando $b=0$)}
        
        Sia $b \in \mathbb{R}^n$ e $\tau_b: \mathbb{R}^n \to \mathbb{R}^n$ $x \mapsto x+b$ cioè la traslazione di vettore $b$ allora 
        $\tau_b$ è differenziabile e $\left(d\tau_b\right)_{x_0} = id_{\mathbb{R}^n}$ per ogni $x_0 \in \mathbb{R}^n$

        Perchè $\displaystyle \tau_b(x)-\tau_b(x_0)-id_{\mathbb{R}^n}(x-x_0) =x+b-(x_0+b)-(x-x_0)=0 \Rightarrow$
        
        $\displaystyle \Rightarrow \lim_{x \to x_0} \frac{\tau_b(x)-\tau_b(x_0)-id_{\mathbb{R}^n}(x-x_0)}{\norm{x-x_0}} = 0$
        \end{esempio}
        \begin{osservazione}
        La composizione di due applicazioni differenziabili è differenziabile e vale:
            \begin{equation}
            \label{eq:composizione_di_applicazioni_differenziabili}
            \left(d(G \circ F)\right)_{x_0} = \left(dG\right)_{F(x_0)} \circ \left(dF\right)_{x_0}
            \end{equation}
        \end{osservazione}
        \begin{definizione}
        $R: \mathbb{R}^n \to \mathbb{R}^n$ si dice \textcolor{red}{isometria lineare} (o operatore unitario) se $R$ è lineare e conserva il prodotto scalare cioè
        \[ \forall x,y \in \mathbb{R}^n \quad R(x) \cdot R(y) = x \cdot y \]
        o equivalentemente conserva le norme 
        \[ \forall x \in \mathbb{R}^n \quad \norm{R(x)} = \norm{x} \]
        \end{definizione}
        \begin{proposizione}\textcolor{violet}{(vista in Geo $2$)}
        \label{prop:caratterizzazione_isometria_lineare}

        $R: \mathbb{R}^n \to \mathbb{R}^n$ è un'isometria lineare $\iff
        \begin{cases} 
           R(0_{\mathbb{R}^n})=0_{\mathbb{R}^n} \\
            \forall x,y \in \mathbb{R}^n \quad \norm{R(x)-R(y)}=\norm{x-y}
        \end{cases}$

        \textcolor{violet}{(In Geometria $2$ l'abbiamo vista in generale per ogni spazio vettoriale euclideo non solo $\mathbb{R}^n$ e ci dice
        che le isometrie lineari conservano le distanze)}
        \end{proposizione}
        \begin{definizione}
        $\phi: \mathbb{R}^n \to \mathbb{R}^n$ si dice \textcolor{red}{isometria} se $\forall x,y \in \mathbb{R}^n$ 
        $\norm{\phi(x)-\phi(y)} = \norm{x-y}$ \textcolor{violet}{(da non confondere con la precedente definizione di isometria lineare questa conserva solo le distanze tra i punti 
        quindi la prop \ref{prop:caratterizzazione_isometria_lineare} ci dice che le isometrie lineari in $\mathbb{R}^n$ sono tutte e sole le isometrie che mandano il vettore
        nullo nel vettore nullo)}
        \end{definizione}
        \begin{proposizione}
        \label{prop:caratterizzazione_isometria}
        Sia $\phi: \mathbb{R}^n \to \mathbb{R}^n$ un'isometria allore $\exists \, ! R: \mathbb{R}^n \to \mathbb{R}^n$ isometria lineare e
        $\exists \, ! b \in \mathbb{R}^n$ tale che $\phi = \tau_b \circ R$ \textcolor{violet}{(cioè $\phi(x) = R(x)+b$ ogni isometria si scrive come composizione
        di una traslazione e un'isometria lineare)}
        \end{proposizione}
        \begin{proof}
        Supponiamo che $\exists R: \mathbb{R}^n \to \mathbb{R}^n$ isometria lineare e $\exists b \in \mathbb{R}^n$ tale che $\phi = \tau_b \circ R$ 

        $\phi(0_{\mathbb{R}^n}) = \tau_b(R(0_{\mathbb{R}^n})) \overset{\text{prop.} \ref{prop:caratterizzazione_isometria_lineare}}{=} \tau_b(0_{\mathbb{R}^n}) = b \Rightarrow b = \phi(0_{\mathbb{R}^n})$
        
        Sia \textcolor{violet}{(si ottiene componendo a destra con l'inversa di $\tau_b$)} $R= \tau_{-b} \circ \phi$ verifichiamo 
        che $R$ è un'isometria lineare, $R$ è isometria in quanto composta di isometrie \textcolor{violet}{(la traslazione inversa e $\phi$)} e inoltre

        $R(0_{\mathbb{R}^n})= \tau_{-b}(\phi(0_{\mathbb{R}^n})) = \tau_{-b}(b) = 0_{\mathbb{R}^n} \overset{\text{prop.} \ref{prop:caratterizzazione_isometria_lineare}}{\Rightarrow} R$ è isometria lineare.
        
        \textcolor{violet}{In questa dimostrazione abbiamo supposto che esistano $R$ e $b$ tali che $\phi = \tau_b \circ R$ e abbiamo mostrato come determinarli esplicitamente.}
        \end{proof}
        \begin{osservazione}
        Se $\phi: \mathbb{R}^n \to \mathbb{R}^n$ è un'isometria allora $\phi$ è differenziabile e per la
        
        proposizione \ref{prop:caratterizzazione_isometria} $\phi= \tau_b \circ R$ 
        
        $\Rightarrow \left(d\phi\right)_{x_0} = \left(d \left( \tau_b \circ R \right) \right)_{x_0} \overset{\eqref{eq:composizione_di_applicazioni_differenziabili}}{=} \left(d\tau_b\right)_{R(x_0)} \circ \left(dR\right)_{x_0} = id_{\mathbb{R}^n} \circ R = R$ 
        \end{osservazione}

        \begin{definizione}
        Sia $F:A \to \mathbb{R}^m$ con $A \subseteq \mathbb{R}^n$ aperto si definiscono le \textcolor{red}{funzioni componenti} di $F$ 

        \textcolor{red}{$F^i = \pi^i \circ F : A \to \mathbb{R}^n$} per $i=1,\ldots,m$ dove $\pi^i: \mathbb{R}^m \to \mathbb{R}$ è la proiezione 
        $i$-esima 

        Se $F$ è differenziabile in $\displaystyle x_0 \in A \overset{\textcolor{violet}{\substack{\text{dim in} \\ \text{analisi 3} }}}{\Rightarrow} \forall i=1,\ldots,m \quad \forall j=1,\ldots,n \quad$ \scalebox{1.6}{$\exists$} $\displaystyle \frac{\partial F^i}{\partial x_j}(x_0)$ \textcolor{blu}{(le $F^i$ 
        sono parzialmente differenziabili in $x_0$)}
        
        Si definisce \textcolor{red}{matrice Jacobiana di $F$ in $x_0$} la matrice 
        \[ \textcolor{red}{\text{\scalebox{1.4}{\text{$J_F(x_0)$}}}} = \begin{pmatrix} \frac{\partial F^1}{\partial x_1}(x_0) & \ldots & \frac{\partial F^1}{\partial x_n}(x_0) \\
        \vdots & \ddots & \vdots \\
        \frac{\partial F^m}{\partial x_1}(x_0) & \ldots & \frac{\partial F^m}{\partial x_n}(x_0) \end{pmatrix} = \left( \frac{\partial F^i}{\partial x_j}(x_0) \right)_{\substack{1 \leq i \leq m \\ 1 \leq j \leq n}} \in \scalebox{1.4}{\text{$M_{m,n}(\mathbb{R})$}} \]
        La matrice $J_F(x_0)$ è la matrice associata a $\left(dF\right)_{x_0}:R^n \to R^m$ rispetto alle basi canoniche di $\mathbb{R}^n$ e $\mathbb{R}^m$ 
        dove $\left\{e_j\right\}_{j=1,\ldots,n}$ è la base canonica di $\mathbb{R}^n$ e
        $\left\{e'_i\right\}_{i=1,\ldots,m}$ è la base canonica di $\mathbb{R}^m$
        
        $\displaystyle dF_{x_0}(e_j) = \sum_{i=1}^m \frac{\partial F^i}{\partial x_j}(x_0) e'_i \overset{\textcolor{violet}{\substack{j\text{-esima} \\ \text{colonna della} \\ \text{Jacobiana}}}}{=} \begin{pmatrix}
        \frac{\partial F^1}{\partial x_j}(x_0) \\
        \vdots \\
        \frac{\partial F^m}{\partial x_j}(x_0)
        \end{pmatrix}$
        
        Si dice che $F:A \to \mathbb{R}^m$ è \textcolor{red}{differenziabile di classe $C^1$} se ammette derivate parziali in ogni punto di $A$ e 
        $\displaystyle \frac{\partial F^1}{\partial x_j}:A \to \mathbb{R}$ sono continue.

        Si dice che $F:A \to \mathbb{R}^m$ è \textcolor{red}{differenziabile di classe $C^k$} se ammette derivate parziali continue di ordine $k$ in ogni punto di $A$

        Si dice che $F:A \to \mathbb{R}^m$ è \textcolor{red}{differenziabile di classe $C^\infty$} se è di classe $C^k \quad \forall k \geq 1$ \textcolor{violet}{(in questo corso
        lavoreremo solo con funzioni di classe $C^\infty$ quindi ogni volta che diremo applicazione differenziabile ci riferiremo a una funzione di classe $C^\infty$)} una funzione di classe $C^\infty$ è detta anche \textcolor{red}{liscia} o \textcolor{red}{smooth}.
        \end{definizione}
    \section{Spazio tangente e fibrato di \texorpdfstring{$\mathbb{R}^n$}{R\textasciicircum n}}
        \begin{definizione}
        Sia $p \in \mathbb{R}^n$ definiamo \textcolor{violet}{(questa definizione sarà coerente con la nozione di spazio tangente in varietà differenziabili)}
        \[ \textcolor{red}{T_p \mathbb{R}^n := \left\{p\right\} \times \mathbb{R}^n = \left\{x_p=(p,x) \mid x \in \mathbb{R}^n \right\}} \]
        $T_p \mathbb{R}^n$ possiende struttura di spazio vettoriale reale mediante le leggi di composizione
        \[ x_p + y_p := (p,x) + (p,y) = (p,x+y) \]
        \[ \alpha x_p := \alpha (p,x) = (p, \alpha x) \quad \alpha \in \mathbb{R}\]
        In particolare $p_2: T_p \mathbb{R}^n \to \mathbb{R}^n$ $p_2(x_p) = p_2(p,x) = x$ è un isomorfismo di spazi vettoriali 
        \textcolor{violet}{($\Rightarrow$ la dimensione di $T_p \mathbb{R}^n$ è $n$)}.

        $T_p \mathbb{R}^n$ è detto \textcolor{red}{spazio tangente in $p$ a $\mathbb{R}^n$}
        \end{definizione}
        \begin{definizione}
        Sia $A \subset \mathbb{R}^n$ aperto, $F: A \to \mathbb{R}^m$ continua e differenziabile $p \in A$ e 
        
        $\left(dF\right)_p: \mathbb{R}^n \to \mathbb{R}^m$ mediante $F$
        e $\left(dF\right)_p$ si può definire l'applicazione 
        \[ T_p \mathbb{R}^n \to T_{F(p)} \mathbb{R}^m \quad (p,x) \mapsto (F(p), \left(dF\right)_p(x)) \]
        Tale applicazione sarà denotata ancora \textcolor{violet}{(abuso di notazione)} con \textcolor{red}{$\left(dF\right)_p$} e denominata \textcolor{red}{differenziale di $F$ in $p$}.
        
        \textcolor{violet}{In alcune circostanze sarà utile pensare il vettore come vettore applicato in $p$ mandato in un vettore $F(p)$}
        \end{definizione}
        \begin{definizione}
        Sia $A \subset \mathbb{R}^n$ aperto, poniamo 
        \[ T A := \bigcup_{p \in A} T_p \mathbb{R}^n \]
        unione disgiunta degli spazi tangenti a $\mathbb{R}^n$ nei punti di $A$.
        \begin{align*}
        X \in T A &\iff \exists ! \, p \in A \text{ tale che } X \in T_p \mathbb{R}^n=\left\{p\right\} \times \mathbb{R}^n  \\
        &\iff \exists ! \, p \in A \text{ tale che } X=(p,x) \text{ con } x \in \mathbb{R}^n
        \end{align*}
        Si considera l'applicazione 
        $\uppi: T A \to A \quad X \mapsto p $
        
        La terna \textcolor{red}{$(T A, \uppi, A)$} è detta \textcolor{red}{fibrato tangente di $A$} 

        $\uppi$ è detta \textcolor{red}{proiezione del fibrato}

        $TA$ è detto \textcolor{red}{spazio totale del fibrato}

        $A$ è detto \textcolor{red}{spazio base}

        $\forall p \in A \quad \uppi^{-1}(p) = T_p \mathbb{R}^n$ è detta \textcolor{red}{fibra in $p$}
        \end{definizione}
    \section{Curve parametrizzate in \texorpdfstring{$\mathbb{R}^n$}{R\textasciicircum n}}
        \begin{definizione}
        Sia $I \subseteq \mathbb{R}$ un intervallo aperto si definisce \textcolor{red}{curva parametrizzata di $\mathbb{R}^n$} 
        ogni applicazione $c: I \to \mathbb{R}^n$ che sia $C^{\infty}$ differenziabile, \textcolor{red}{$t$} si dice \textcolor{red}{parametro} della curva.

        L'insieme \textcolor{red}{$\left\{c(t) \mid t \in I \right\}$} è detto \textcolor{red}{supporto della curva} e sarà denotato con \textcolor{red}{$\supp(c)$}

        Poniamo $\forall t \in I \quad c(t)= (c^1(t),\ldots,c^n(t))$ con $c^i= p^i \circ c: I \to \mathbb{R}$ per $i=1,\ldots,n$ \textcolor{blu}{(funzioni componenti di $c$)}
        
        $\dot{c}(t)= (\dot{c}^1(t),\ldots,\dot{c}^n(t))$

        Per $t_0 \in I$ si può considerare l'applicazione lineare $\left(dc\right)_{t_0}: \mathbb{R} \to \mathbb{R}^n$ 
        \[ \left(dc\right)_{t_0} \textcolor{blu}{\underbrace{\textcolor{black}{(1)}}_{\substack{\text{base} \\ \text{canonica} \\ \text{di } \mathbb{R}}}} \overset{\textcolor{violet}{\substack{\text{unica} \\ \text{colonna} \\ \text{matrice} \\ \text{Jacobiana}}}}{=} \left(\dot{c}^1(t_0), \ldots, \dot{c}^n(t_0) \right) = \dot{c}(t_0) \] 
        Si definisce inoltre \textcolor{violet}{(per abuso di notazione continuo a denotare come differenziale)}
        \[ \left(dc\right)_{t_0}: T_{t_0} \mathbb{R} \to T_{c(t_0)} \mathbb{R}^n \quad \left(t_0,t\right) \mapsto \left(c(t_0), (dc)_{t_0} (t) \right) \]
        $\left\{(t_0,1)\right\}$ è base di $T_{t_0} \mathbb{R}$ 

        Si definisce \textcolor{red}{vettore tangente in $t_0$ a $c$} il vettore
        \[ \textcolor{red}{\left(dc\right)_{t_0} (t_0,1) = \left( c(t_0), (dc)_{t_0}(1) \right) = \left(c(t_0), \dot{c}(t_0) \right)} \, \textcolor{violet}{\equiv \dot{c}(t_0)} \]
        Il vettore tangente in $t_0$ a $c$ può essere denotato con $\dot{c}(t_0)$ \textcolor{violet}{(abuso di notazione)} omettendo il punto di applicazione $c(t_0)$  
        \end{definizione}
        \begin{definizione}
        Una curva parametrizzata $c: I \to \mathbb{R}^n$ si dice \textcolor{red}{regolare} se \textcolor{red}{$\forall t \in I \quad \dot{c}(t) \neq 0$}
        \textcolor{blu}{(Il vettore tangente in ogni punto della curva è non nullo)} 
        \end{definizione}

        \begin{definizione}
        Siano $\alpha: I \to \mathbb{R}^n$ e $\beta: J \to \mathbb{R}^n$ curve parametrizzate regolari con $I,J \subset \mathbb{R}$ intervalli aperti

        Si dice che \textcolor{red}{$\alpha$} e \textcolor{red}{$\beta$} sono \textcolor{red}{equivalenti} \textcolor{blu}{(a meno di parametrizzazioni)} 
        se esiste $\phi: I \to J$ diffeomorfismo \textcolor{blu}{(applicazione differenziabile bigettiva con inversa differenziabile)} tale che \textcolor{red}{$\beta = \alpha \circ \phi$}
        \begin{equation*}
        \begin{tikzcd}[row sep=large, column sep=large]
        J \arrow[r, "\phi"] \arrow[rr, bend right, "\beta"'] & I \arrow[r, "\alpha"] & \mathbb{R}^n
        \end{tikzcd}
        \end{equation*}
        Scriveremo che $\beta \sim \alpha$ e $\phi$ è detto cambiamento di parametro
        \end{definizione}
        \begin{osservazione}
        $\phi$ è differenziabile $C^\infty$ e invertibile $\Rightarrow \phi$ è strettamente monotona 
        
        $\Rightarrow \left( \forall s \in J \quad \phi'(s) > 0 \right) \lor \left( \forall s \in J \quad \phi'(s) < 0 \right)$
        \end{osservazione}
        \begin{definizione}
        Si dice che 

        \textcolor{red}{$\phi$ conserva l'orientazione} se $\forall s \in J \quad \phi'(s) > 0$ 

        \textcolor{red}{$\phi$ rovescia l'orientazione} se $\forall s \in J \quad \phi'(s) < 0$
        \end{definizione}
        \begin{osservazione}
        Supponiamo $\alpha \sim \beta$ tramite $\alpha : I \to \mathbb{R}^n$ e $\beta: J \to \mathbb{R}^n$ e $\beta = \alpha \circ \phi$ con $\phi: J \to I$ diffeomorfismo

        $\beta = \alpha \circ \phi \Rightarrow \supp (\alpha) = \supp(\beta)$
        \begin{proof}
        $\supp(\beta)= \beta(J)= \left(\alpha \circ \phi \right)(J) = \alpha(I)= \supp(\alpha)$
        \end{proof}
        $\beta = \alpha \circ \phi \Rightarrow \forall s \in J \quad \dot{\beta}(s) = \phi'(s) \cdot \dot{\alpha}(\phi(s)) $
        \begin{proof}
        \begin{align*}
        \hat{\beta}(s) &= \left( d \beta \right)_s (1) = \left( \alpha \circ \phi \right)_s (1) \overset{\eqref{eq:composizione_di_applicazioni_differenziabili}}{=} \left(d\alpha\right)_{\phi(s)} \circ \left(d\phi\right)_s (1) \\
        &= \left(d\alpha\right)_{\phi(s)} \left( \phi'(s) \right)  = \left(d\alpha\right)_{\phi(s)} \left( \phi'(s) \cdot 1 \right) = \phi'(s) \cdot \left(d\alpha\right)_{\phi(s)} (1)  \\
        &= \phi'(s) \cdot \dot{\alpha}(\phi(s)) 
        \end{align*}
        \end{proof}
        \end{osservazione}

        \begin{definizione}
        La relazione di "equivalenza a meno di parametrizzazione" è una relazione di equivalenza (simmetrica , riflessiva e transitiva) sull'insieme
        delle curve parametrizzate regolari.

        Ogni classe di equivalenza \textcolor{red}{$\left[ \alpha \right]_{\sim}$} è detta \textcolor{red}{curva non parametrizzata} 
        \end{definizione}
        \begin{definizione}
        Si dice che $2$ curve parametrizzate regolari $\alpha: I \to \mathbb{R}^n$ e $\beta: J \to \mathbb{R}^n$ sono \textcolor{red}{equivalenti con orientazione} se 
        esiste un cambiamento di parametro $\phi$ che conservi l'orientazione.
        
        Scriveremo \textcolor{red}{$\alpha \sim_0 \beta$} e anche questa è una relazione di equivalenza.
        
        Ogni classe di equivalenza \textcolor{red}{$\left[\alpha\right]_{\sim_0}$} è detta \textcolor{red}{curva non parametrizzata orientata}

        \textcolor{violet}{Vedremo delle proprietà invarianti per cambiamento di parametro}
        \end{definizione}

        \begin{definizione}
        Sia $c: I \to \mathbb{R}^n$ curva parametrizzata regolare, sia $[a,b] \subset I$ e si definisce \textcolor{red}{lunghezza dell'arco di curva $c_{\restriction{[a,b]}}$}
        \begin{equation}
        \label{eq:lunghezza_arco_di_curva}
        L \left(c_{\restriction{[a,b]}}\right)= \int_a^b \norm{\dot{c}(t)} \, dt
        \end{equation}
        \end{definizione}
        \begin{proposizione}
        \label{prop:lunghezza_arco_di_curva_invariante}
        La lunghezza di un arco di curva è invariante per cambiamento di parametro
        \end{proposizione}
        \begin{proof}
        Siano $\alpha:I \to \mathbb{R}^n \quad \beta : J \to \mathbb{R}^n$ curve parametrizzate regolari equivalenti, sia $\phi : J \to I$ diffeomorfismo tale che $\beta = \alpha \circ \phi$

        Sia $[a,b] \subset J$ e verifichiamo che se $\phi$ conserva l'orientazione $\Rightarrow L \left(\beta_{\restriction{[a,b]}}\right)= L\left(\alpha_{\restriction{[\phi(a),\phi(b)]}}\right)$ e
        che se $\phi$ rovescia l'orientazione $\Rightarrow L \left(\beta_{\restriction{[a,b]}}\right)= L\left(\alpha_{\restriction{[\phi(b),\phi(a)]}}\right)$ 
        
        Supponiamo ora che $\phi$ conservi l'orientazione
        \begin{align*}
        L(\beta_{\restriction{[a,b]}}) =& \int_a^b \norm{\dot{\beta}(s)} \, ds = \int_a^b \norm{ \phi'(s) \dot{\alpha}(\phi(s))} \, ds \\
        =& \int_a^b \left|\phi'(s)\right| \, \norm{\dot{\alpha}(\phi(s))} \, ds \overset{\textcolor{blu}{\substack{\text{Supponiamo che} \\ \phi \text{ conservi} \\ \text{l'orientazione}}}}{=} \\
        =& \int_a^b  \phi'(s) \, \norm{\dot{\alpha}(\phi(s))} \, ds \quad \text{ posto } t=\phi(s) \quad dt= \phi'(s) ds \\
        =& \int_{\phi(a)}^{\phi(b)} \norm{\dot{\alpha} (t)} \, dt = L\left(\alpha_{\restriction{[\phi(a),\phi(b)]}}\right)
        \end{align*}
        Supponiamo ora che $\phi$ rovesci l'orientazione
        \begin{align*}
        L(\beta_{\restriction{[a,b]}}) =& \int_a^b \norm{\dot{\beta}(s)} \, ds = \int_a^b \norm{ \phi'(s) \dot{\alpha}(\phi(s))} \, ds \\
        =& \int_a^b \left|\phi'(s)\right| \, \norm{\dot{\alpha}(\phi(s))} \, ds \overset{\textcolor{blu}{\substack{\text{Supponiamo che} \\ \phi \text{ rovesci} \\ \text{l'orientazione}}}}{=} \\
        =& - \int_a^b  \phi'(s) \, \norm{\dot{\alpha}(\phi(s))} \, ds \quad \text{ posto } t=\phi(s) \quad dt= \phi'(s) ds \\
        =& \int_{\phi(b)}^{\phi(a)} \norm{\dot{\alpha} (t)} \, dt = L\left(\alpha_{\restriction{[\phi(b),\phi(a)]}}\right)
        \end{align*}
        \end{proof}
        \begin{definizione}
        Una curva parametrizzata $c: I \to \mathbb{R}^n$ si dice \textcolor{red}{parametrizzata per lunghezza d'arco} se \textcolor{red}{$\forall t \in I \quad \norm{\dot{c}(t)}=1$} 
        \textcolor{blu}{(il vettore tangente in ogni punto è unitario)}
        \end{definizione}
        \begin{osservazione}
        Se $c$ è una curva parametrizzata per lunghezza d'arco $t_0 >0$
        \[ L \left(c_{\restriction{[a,t_0]}}\right) = \int_a^{a+t_0} \norm{\dot{c}(t)} \, dt = a+t_0 - a = t_0 \]
        \end{osservazione}
        \begin{proposizione}
        \label{prop:esistenza_riparametrizzazione_lunghezza_arco}
        Ogni curva parametrizzata regolare ammette una riparametrizzazione per lunghezza d'arco \textcolor{blu}{(equivalente)}
        \end{proposizione}
        \begin{proof}
        Sia $\alpha : I \to \mathbb{R}^n$ curva regolare sia $t_0 \in I$

        Sia \textcolor{red}{$\displaystyle \psi:I \to R \quad \psi(t)=\int_{t_0}^t \norm{\dot{\alpha}(s)} \, ds$} $\psi$ è continua e derivabile
        $\psi'(t) = \norm{\dot{\alpha}(t)} \overset{\textcolor{orange}{\text{regolarità}}}{>} 0 \Rightarrow \psi$ è strettamente crescente $\Rightarrow \psi$ è ingettiva

        $\psi: I \to \psi(I)=J$ è bigettiva e $\phi= \psi^{-1}:J \to I$ è diffeomorfismo

        $\displaystyle \forall s \in J \quad \phi'(s) \overset{\textcolor{violet}{\substack{\text{derivazione} \\ \text{funzione} \\ \text{inversa}}}}{=} 
        \frac{1}{\psi'(\phi(s))} = \frac{1}{\norm{\dot{\alpha}(\phi(s))}}>0$

        Sia \textcolor{red}{$\beta = \alpha \circ \phi:J \to \mathbb{R}^n$} verifico che $\beta$ è parametrizzata per lunghezza d'arco
        \begin{align*}
        \norm{\dot{\beta}(s)} =& \left|\phi'(s)\right| \, \norm{\dot{\alpha} (\phi(s))} \\
        =& \phi'(s) \, \norm{\dot{\alpha} (\phi(s))} = \frac{1}{\norm{\dot{\alpha}(\phi(s))}} \, \norm{\dot{\alpha} (\phi(s))} = 1
        \end{align*}
        \end{proof}

        \begin{definizione}
        Sia $\mathcal{C} \subseteq \mathbb{R}^n$ e si consideri su $\mathcal{C}$ la topologia indotta dalla topologia euclidea
        \textcolor{blu}{($U_{\mathcal{C}} \subseteq \mathcal{C}$ è aperto $\iff U_{\mathcal{C}} = U \cap \mathcal{C}$ con $U \subseteq \mathbb{R}^n$ aperto)}
        
        Si dice che $\mathcal{C}$ è una \textcolor{red}{curva differenziabile} se $\forall p \in \mathcal{C}$ esiste \textcolor{red}{$\alpha : I_{\alpha} \to U_{\alpha} \subset \mathcal{C} \subset \mathbb{R}^n$ 
        omeomorfismo} tra $I_{\alpha}$ intervallo aperto di $\mathbb{R}$ e $U_{\alpha}$ aperto di $\mathcal{C}$ tale che $p \in U_{\alpha}$

        $\alpha$ è una curva parametrizzata regolare e $\alpha$ è detta \textcolor{red}{parametrizzazione locale} della curva.

        Si osservi che il supporto della parametrizzazione $\alpha$ è $\alpha(I_{\alpha})=U_{\alpha}$

        Posto $\varphi_{\alpha} := \alpha^{-1} : U_{\alpha} \subset \mathcal{C} \to I_{\alpha} \subset \mathbb{R}$ la coppia 
        \textcolor{red}{$\left(U_{\alpha}, \varphi_{\alpha}\right)$} è detta \textcolor{red}{carta locale} nel punto $p$ dalla definizione segue che una curva differenziabile 
        ammette una famiglia di carte locali \textcolor{red}{$\{(U_{\alpha}, \varphi_{\alpha})\}_{\alpha \in A}$} tale che $\displaystyle \mathcal{C}=\bigcup_{\alpha \in A} U_{\alpha}$
        tale famiglia è detta \textcolor{red}{atlante differenziabile}

        \begin{proposizione}
        \label{prop:funzione_di_transizione_curve}
        Sia $\mathcal{C} \subset \mathbb{R}^n$ curva differenziabile siano $\alpha: I_\alpha \to U_\alpha$ e $\beta: I_\beta \to U_\beta$ due parametrizzazioni 
        locali di $\mathcal{C}$. Se $U_\alpha \cap U_\beta \neq \emptyset$, allora l'applicazione 
        
        $\textcolor{red}{\beta^{-1} \circ \alpha}: \alpha^{-1} \left(U_\alpha \cap U_\beta \right) \to \beta^{-1} \left(U_\alpha \cap U_\beta \right)$ 
        \textcolor{blu}{$\left(\equiv  \varphi_\beta \circ \varphi_\alpha^{-1} : \varphi_{\alpha} \left(U_\alpha \cap U_\beta \right) \to \varphi_\beta \left(U_\alpha \cap U_\beta \right) \right)$} è 
        
        un \textcolor{red}{diffeomorfismo}
        \end{proposizione}
        \begin{proof}
        Sia $t_0 \in \alpha^{-1} \left(U_{\alpha} \cap U_{\beta} \right) \subset I_\alpha$
        
        Verifichiamo che $\beta^{-1} \circ \alpha$ è differenziabile in un introno di $t_0$

        Posto $p = \alpha(t_0) \in U_{\alpha} \cap U_{\beta}$ sia $s_0 = \beta^{-1}(p) \in \beta^{-1} \left(U_{\alpha} \cap U_{\beta} \right) \subset I_\beta$
        $\Rightarrow \alpha(t_0)= \beta(s_0)=p$

        Poniamo $\forall s \in I_\beta \quad \beta(s)=\left(x^1(s), \ldots, x^n(s)\right)$ ovvero $x^i(s)=p^i(\beta(s))$ 
        con $p^i: \mathbb{R}^n \to \mathbb{R}$ proiezione

        $\dot{\beta}(s) = \left(\dot{x}^1(s), \ldots, \dot{x}^n(s)\right)$
        
        $\beta$ è regolare $\Rightarrow \dot{\beta}(s_0) \neq 0 \Rightarrow \exists i \in \{1, \ldots, n\}$ tale che $\dot{x}^i(s_0) \neq 0$

        La funzione $s \in \beta^{-1} \left(U_{\alpha} \cap U_{\beta} \right) \subset I_\beta \mapsto x^i(s)= p_i(\beta(s)) \in \mathbb{R}$ 
        è differenziabile.

        Per il teorema della funzione inversa $\exists J $ intorno aperto di $s_0 \quad J \subset \beta^{-1} \left(U_{\alpha} \cap U_{\beta} \right)$ ed
        $\exists V$ intorno aperto di $x^i(s_0) = p_i(\beta(s_0)) = p_i(p)$ tale che $p_i \circ \beta: J \to V$ sia un diffeomorfismo.

        $p_i^{-1}(V) \cap \mathcal{C}$ è un aperto di $\mathcal{C}$ che contiene $\beta(s_0)=\alpha(t_0)=p$.

        Per la continuità di $\alpha$ esiste un intervallo aperto $I \subset \alpha^{-1} \left(U_\alpha \cap U_\beta \right)$ tale che
        $t_0 \in I$
        
        $\alpha(I) \subset p_i^{-1}(V) \cap \mathcal{C}$
        \[ I \xrightarrow[]{\quad p_i \circ \alpha \quad} V \xrightarrow[]{\quad \left(p_i \circ \beta \right)^{-1} \quad} J \]
        Questa composizione è differenziabile 

        $\forall t \in I \quad \left(p_i \circ \beta \right)^{-1} \left( \left( p_i \circ \alpha \right) (t) \right) = \beta^{-1} \left( \alpha (t) \right)$
        \textcolor{blu}{(perché $\left(p_i \circ \beta \right) \left( \beta^{-1}(\alpha(t)) \right)) = \left(p_i \circ \alpha \right) (t)$)}

        Pertanto $\beta^{-1} \circ \alpha : I \to J$ è differenziabile.

        Scambiando i ruoli di $\alpha$ e $\beta$ risulta che $\alpha^{-1} \circ \beta$ è differenziabile.

        Pertanto $\beta^{-1} \circ \alpha : \alpha^{-1} \left(U_{\alpha} \cap U_{\beta} \right) \to \beta^{-1} \left(U_{\alpha} \cap U_{\beta} \right)$ è diffeomorfismo.
        \end{proof}
        \begin{osservazione}
        Posto $\phi = \beta^{-1} \circ \alpha \quad \alpha = \beta \circ \phi$ con $\phi$ diffeomorfismo quindi le $2$ curve parametrizzate sono equivalenti. 
        \end{osservazione}

    \section{Riferimenti di Frenet}
        \begin{definizione}
        Sia $c: I \to \mathbb{R}^n$ una curva parametrizzata si dice \textcolor{red}{$n$-riferimento mobile lungo $c$} una famiglia di applicazioni
        differenziabili \textcolor{red}{$e_i: I \to \mathbb{R}^n$} per $i=1,\dots,n$ tali che 
        
        $\forall t \in I \quad \forall i,j \in \{1,\dots,n\} \quad$ \textcolor{red}{$e_i(t) \cdot e_j(t) = \delta_{ij}$}
        \textcolor{blu}{(ovvero $\forall t \in I \quad (e_i(t))_{i=1,\dots,n}$ è base ortonormale di $\mathbb{R}^n$)}
        Identificando $\mathbb{R}^n \equiv T_{c(t)}\mathbb{R}^n \quad \left(e_i(t)\right)_{1 \leq i \leq n}$ può 
        essere riguardata come una famiglia di campi vettoriali differenziabili lungo $c$ \textcolor{blu}{($e_i(t) \equiv (c(t), e_i(t)) \in T_{c(t)}\mathbb{R}^n$)}.
        \begin{center}
        \begin{tikzpicture}[scale=1.5]

        % 1. Intervallo I
        \draw[thick] (0, 0) -- (1.5, 0);
        \draw[thick] (0, -0.15) -- (0, 0.15);
        \draw[thick] (1.5, -0.15) -- (1.5, 0.15);
        \node[below=4pt] at (0.75, 0) {\Large $I$};

        % 2. Curva parametrizzata in R^2 (supporto)
        \draw[very thick, teal] (-0.5, 1.2)

            node[right, text=black] {\Large $\mathbb{R}^2$};

        % 3. Mappa c dall'intervallo alla curva
        \draw[->, thick, teal] (0.8, 0.3) to[bend left=35] node[midway, left] {\large $c$} (1, 1.6);

        % 4. Primo punto c(t) e i suoi vettori ortonormali
        \coordinate (P1) at (1.2, 2);
        \fill[green!70!black] (P1) circle (1.5pt) node[below=3pt] {$c(t_1)$};

        \coordinate (P2) at (3.5, 1.5);

        \end{definizione}
        \begin{definizione}
        Un $n$-riferimento mobile lungo $c$ $\left(e_i\right)_{1 \leq i \leq n}$ si dice \textcolor{red}{riferimento di Frenet}
        se $\forall k \leq n-1 \quad \forall t \in I \quad c^{(k)}(t) \in \langle e_1(t), \dots, e_k(t)\rangle$ \textcolor{blu}{($c^{(k)}$ denota il vettore
        derivato di $c$ $k$-volte)}
        \end{definizione}
        \begin{esempio}
        $c_I \to \mathbb{R}^2$ $(e_1, e_2)$ riferimento mobile lungo $c$ 

        $(e_1, e_2)$ è un riferimento di Frenet lungo $c \iff \forall t \in I \quad c'(t) \in \langle e_1(t) \rangle$
        
        \textcolor{violet}{In altre parole, il primo vettore del riferimento di Frenet è sempre tangente alla curva.}
