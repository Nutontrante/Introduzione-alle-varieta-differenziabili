    \section{Superfici parametrizzate \texorpdfstring{$\mathbb{R}^3$}{R\textasciicircum 3}}
        \begin{definizione}
        Sia $A \subset \mathbb{R}^2$ aperto connesso si definisce \textcolor{red}{superficie parametrizzata in $\mathbb{R}^3$} 
        ogni applicazione $\textcolor{red}{f \colon A \to \mathbb{R}^3}$ differenziabile di classe $C^\infty$ 
        \textcolor{violet}{(parleremo solo di differenziabilità $C^\infty$)}.

        L'insieme $f(A) \subset \mathbb{R}^3$ è detto \textcolor{red}{supporto di $f$}

        \textcolor{violet}{Se $f$ è un'applicazione differenziabile da un aperto $A \subset \mathbb{R}^2$ in $\mathbb{R}^3$ i ogni 
        punto il suo differenziale è una applicazione lineare $ \left(df\right)_u \colon \mathbb{R}^2 \to \mathbb{R}^3$ }

        Si dice che $f$ è \textcolor{red}{regolare} se $\forall u \in A \quad \textcolor{red}{\left(df\right)_u \colon \mathbb{R}^2 \to \mathbb{R}^3}$ \textcolor{red}{è ingettiva}.

        Siano $f^i = p_i \circ f \colon A \to \mathbb{R}$ le funzioni componenti di $f$ tali che $\forall u \in A \quad f(u) = \left(f^1(u), f^2(u), f^3(u)\right)$.

        La matrice Jacobiana di $f$ in $u \in A$ è
        \[ \left(Jf\right)_u = \left( \dfrac{\partial f^{\, i}}{\partial u^{\, j}} \right)_{\substack{i=1,2,3 \\ j=1,2}}= \begin{pmatrix}
        \frac{\partial f^1}{\partial u^1} & \frac{\partial f^1}{\partial u^2} \\[2mm]
        \frac{\partial f^2}{\partial u^1} & \frac{\partial f^2}{\partial u^2} \\[2mm]
        \frac{\partial f^3}{\partial u^1} & \frac{\partial f^3}{\partial u^2}
        \end{pmatrix} \]
        $\left(Jf\right)_u $ è la matrice associata a $\left(df\right)_u \colon \mathbb{R}^2 \to \mathbb{R}^3$ rispetto alle basi canoniche di 
        $\mathbb{R}^2$ e $\mathbb{R}^3$.
        \begin{align*}
        f \text{ è regolare} &\iff \forall u \in A \quad \left(df\right)_u \text{ è ingettiva} \\
        &\iff \ker \left(df\right)_u = \{0\} \iff \dim \left(\im \left(df\right)_u \right) = 2 \\
        &\iff \rg \left( \left(Jf\right)_u  \right) = 2
        \end{align*}
        \end{definizione}

        \begin{definizione}
        Sia  $A \subset \mathbb{R}^2$ aperto connesso e $f \colon A \to \mathbb{R}^3$ superficie regolare.

        Sia $u \in A \quad \left(df\right)_u \colon \mathbb{R}^2 \to \mathbb{R}^3$ ingettiva 

        Definiamo l'applicazione 
        \[ \color{red} \left(df\right)_u \colon T_u \mathbb{R}^2 \to T_{f(u)} \mathbb{R}^3 \quad \left(u,x\right) \mapsto \left(f(u), \left(df\right)_u(x)\right) \]
        questa applicazione lineare è ingettiva 
        
        Si definisce \textcolor{red}{spazio tangente in $u$ a $f$} lo spazio vettoriale 
        \[ \color{red} T_u f:= \left(df\right)_u \left(T_u \mathbb{R}^2 \right) \subset T_{f(u)} \mathbb{R}^3 \]
        $T_u f$ è un sottospazio vettoriale di $T_{f(u)} \mathbb{R}^3$ di dimensione $2$
        
        Sia $\left\{e_1, e_2\right\}$ la base canonica di $\mathbb{R}^2$ $\left\{ \left(u, e_1\right), \, \left(u, e_2\right) \right\}$ è
        base di $T_u \mathbb{R}^2$, una base di $T_u f$ è data da 
        $\left\{\left(df\right)_u \left(u, e_1\right), \, \left(df\right)_u \left(u, e_2\right) \right\}$

        Porremo per $i=1,2$
        \begin{align*}
        f_{u^i} &= \left(df\right)_u \left(u, e_i\right) = \left(f(u), \, \left(df\right)_u \left(e_i\right)\right) \\
        &= \left(f(u), \textcolor{blu}{\underbrace{ \textcolor{black}{\left( \frac{\partial f^1}{\partial u^i}(u), \, \frac{\partial f^2}{\partial u^i}(u), \, \frac{\partial f^3}{\partial u^i}(u) \right)}}_{\substack{\text{prima o seconda colonna} \\ \text{di } (Jf)_u \text{ a seconda} \\ \text{che } i=1,2}}} \right)
        \end{align*}
        Pertanto $T_u f = \left(df\right)_u \left(T_u \mathbb{R}^2 \right) = \langle f_{u^1}, f_{u^2} \rangle \subset T_{f(u)} \mathbb{R}^3$.
        \end{definizione}

        \begin{definizione}
        Sia $f \colon A \to \mathbb{R}^3$ superficie parametrizzata regolare si definisce \textcolor{red}{campo di vettori lungo $f$} 
        ogni applicazione \textcolor{red}{$\hat{X} \colon A \to T \mathbb{R}^3$} tale che \textcolor{red}{$\forall u \in A \quad \hat{X}(u) \in T_{f(u)} \mathbb{R}^3$}

        Pertanto $\hat{X}(u) = \left(f(u), X(u)\right)$ resta quindi definita l'applicazione $X \colon A \to \mathbb{R}^3$ che $u \mapsto X(u)$.

        Se \textcolor{red}{$X: A \to \mathbb{R}^3$ è differenziabile} si dice che \textcolor{red}{$\hat{X}$ è} un \textcolor{red}{campo di vettori differenziabile}

        Nel seguito si potrà identificare $X$ con $\hat{X}$.
        \end{definizione}
        \begin{osservazione}
        Lo spazio $T_{f(u)} \mathbb{R}^3$ si può munire del prodotto scalare definito da 
        \[ \left(f(u), x\right) \cdot \left(f(u), y\right) = x \underbrace{\cdot}_{\substack{\text{prodotto} \\ \text{scalare} \\ \text{standard} \\ \text{in } \mathbb{R}^3}} y \qquad x,y \in \mathbb{R}^3 \]
        Pertanto $T_{f(u)} \mathbb{R}^3 = T_u f \oplus \left(T_u f\right)^\perp \quad \dim \left(T_uf\right)=2 \quad \dim \left(\left(T_u f\right)^\perp\right)=1$
        \end{osservazione}
        \begin{definizione}
        \mbox{}

        Si dice che $\hat{X} \colon A \to T \mathbb{R}^3$ è un \textcolor{red}{campo vettoriale tangente lungo $f$} 
        
        se \textcolor{red}{$\forall u \in A \quad \hat{X}(u) \in T_u f$}
        
        Si dice che $\hat{X} \colon A \to T \mathbb{R}^3$ è un \textcolor{red}{campo vettoriale normale lungo $f$} 

        se \textcolor{red}{$\forall u \in A \quad \hat{X}(u) \in \left(T_u f\right)^\perp$}
        \end{definizione}
        \begin{esempio}
        Sia $f \colon A \to \mathbb{R}^3$ superficie parametrizzata regolare, si considerino le applicazioni 
        \[ f_{u^1} \colon A \to \mathbb{R}^3 \quad u \mapsto f_{u^1}(u) = \left( \frac{\partial f^1}{\partial u^1}(u), \frac{\partial f^2}{\partial u^1}(u), \frac{\partial f^3}{\partial u^1}(u) \right) \in T_u f \]
        \[ f_{u^2} \colon A \to \mathbb{R}^3 \quad u \mapsto f_{u^2}(u) = \left( \frac{\partial f^1}{\partial u^2}(u), \frac{\partial f^2}{\partial u^2}(u), \frac{\partial f^3}{\partial u^2}(u) \right) \in T_u f \]
        $f_{u^1}(u)$ e $f_{u^2}(u)$ definiscono due campi vettoriali tangenti lungo $f$.
        \end{esempio}

        \begin{proposizione}
        Sia $A \subset \mathbb{R}^2$ e $f \colon A \to \mathbb{R}^3$ una superficie parametrizzata regolare, ogni campo vettoriale differenziabile 
        tangente $X$ lungo $f$ si può scrivere in un unico modo come:
        \begin{equation}
        \label{eq:campo_vettoriale_tangente_lungo_una_superficie}
        X(u) = a^1(u) f_{u^1}(u) + a^2(u) f_{u^2}(u) 
        \end{equation}
        dove $a^1, a^2 \colon A \to \mathbb{R}$ sono funzioni differenziabili.
        Viceversa ogni coppia di funzioni differenziabili $a^1, a^2 \colon A \to \mathbb{R}$ determina 
        un campo vettoriale differenziabile tangente lungo $f$ dato da \eqref{eq:campo_vettoriale_tangente_lungo_una_superficie}
        \end{proposizione}
        \begin{proof}
        Sia $X \colon A \to \mathbb{R}^3$ campo vettoriale differenziabile tangente lungo $f$. 

        $\forall u \in A \quad X(u) \in T_u f= \langle f_{u^1}(u), f_{u^2}(u) \rangle$ 

        $\Rightarrow \exists \, ! \, a^1(u), a^2(u) \in \mathbb{R} \quad \text{tali che} $
        \begin{equation}
        \label{eq:campo_vettoriale_tangente_lungo_una_superficie_dim_1}
        X(u) = a^1(u) f_{u^1}(u) + a^2(u) f_{u^2}(u)
        \end{equation}
        Verifichiamo che le funzioni $a^1, a^2 \colon A \to \mathbb{R}$ sono differenziabili.

        Moltiplichiamo scalarmene \eqref{eq:campo_vettoriale_tangente_lungo_una_superficie_dim_1} per $f_{u^i}(u)$ per $i=1,2$ 
        \[ \begin{cases*}
        X(u) \cdot f_{u^1}(u) = a^1(u) f_{u^1}(u) \cdot f_{u^1}(u) + a^2(u) f_{u^2}(u) \cdot f_{u^1}(u) \\
        X(u) \cdot f_{u^2}(u) = a^1(u) f_{u^1}(u) \cdot f_{u^2}(u) + a^2(u) f_{u^2}(u) \cdot f_{u^2}(u)
        \end{cases*} \]
        $\Rightarrow \left(a^1(u), \, a^2(u)\right)$ è soluzione di un sistema lineare di $2$ equazioni in $2$ incognite.

        La matrice incompleta del sistema è 
        \[ A(u) = \begin{pmatrix}
        f_{u^1}(u) \cdot f_{u^1}(u) & f_{u^2}(u) \cdot f_{u^1}(u) \\
        f_{u^1}(u) \cdot f_{u^2}(u) & f_{u^2}(u) \cdot f_{u^2}(u)
        \end{pmatrix} \]
        $\det A(u) = \norm{f_{u^1}(u)}^2 \norm{f_{u^2}(u)}^2 - (f_{u^1}(u) \cdot f_{u^2}(u))^2 \overset{\textcolor{violet}{\substack{\text{proprietà} \\ \text{prodotto} \\ \text{vettoriale}}}}{=} \norm{f_{u^1}(u) \times f_{u^2}(u)}^2 >0$

        L'unica soluzione del sistema di Cramer è data da
        \[ a^1(u) = \frac{ \begin{vmatrix}
        X(u) \cdot f_{u^1}(u) & \quad f_{u^2}(u) \cdot f_{u^1}(u) \\
        X(u) \cdot f_{u^2}(u) & \quad f_{u^2}(u) \cdot f_{u^2}(u)
        \end{vmatrix}}{\det A(u)} \quad
        a^2(u) = \frac{ \begin{vmatrix}
        f_{u^1}(u) \cdot f_{u^1}(u) & \quad X(u) \cdot f_{u^1}(u) \\
        f_{u^1}(u) \cdot f_{u^2}(u) & \quad X(u) \cdot f_{u^2}(u)
        \end{vmatrix}}{\det A(u)} \]
        Essendo $X,f, f_{u^1}, f_{u^2}$ differenziabili, anche $a^1$ e $a^2$ sono differenziabili.
        \end{proof}
    \section{Riferimento di Gauss}
        \begin{definizione}
        Sia $A \subset \mathbb{R}^2$ $f \colon A \to \mathbb{R}^3$ una superficie parametrizzata regolare definiamo 

        $\textcolor{red}{N \colon A \to \mathbb{R}^3} \quad \textcolor{red}{\forall u \in A \quad N(u) = \textcolor{blu}{\underbrace{\textcolor{red}{f_{u^1}(u) \times f_{u^2}(u)}}_{\neq 0 \text{ lin. indip.}}}} $

        $N(u) \cdot f_{u^i}(u) = 0 \quad i=1,2 \Rightarrow N(u) \in \langle f_{u^1}(u), f_{u^2}(u) \rangle^\perp =\left( T_u f \right)^\perp$

        $N$ definisce un \textcolor{red}{campo di vettori normale lungo $f$} essendo $f \colon A \to \mathbb{R}^3$ regolare, $f_{u^1}(u) \times f_{u^2}(u) \neq 0$ 

        Pertanto si può definire \textcolor{red}{$n \colon A \to \mathbb{R}^3$} tale che \textcolor{red}{$\forall u \in A \quad n(u)=\dfrac{f_{u^1}(u) \times f_{u^2}(u)}{\norm{f_{u^1}(u) \times f_{u^2}(u)}}$}

        $n$ definisce un campo di vettori normale unitario lungo $f$ detto \textcolor{red}{campo vettoriale di Gauss}
        \[ \norm{n(u)} = 1 \Rightarrow n(u) \in \textcolor{blu}{\underbrace{\textcolor{black}{S^2= \{ x \in \mathbb{R}^3 \mid \norm{x} = 1 \}}}_{\substack{\text{sfera di centro} \\ 0 \text{ e raggio } 1}}} \]
        L'applicazione \textcolor{red}{$n \colon A \to S^2 \subset \mathbb{R}^3 \quad u \mapsto n(u)$} è detta \textcolor{red}{mappa di Gauss}  

        Il riferimento mobile lungo $f$ definito da \textcolor{red}{$\{ f_{u^1}(u), f_{u^2}(u), n(u) \} $} è detto \textcolor{red}{riferimento di Gauss}
